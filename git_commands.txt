git clone <url> - скачивает репозиторий с сервера
git status - Показывает текущее состояние
git add - добавляет файл
git commit - локально сохраняет изменения
git push - отправляет коммиты на сервер
git pull - скачивает обновления с сервера
git branch - список веток
git checkout - переключение с ветки на ветку
git merge - соединяет ветки 
git log - показывает историю коммитов
.
